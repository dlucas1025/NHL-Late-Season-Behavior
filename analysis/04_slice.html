Draft & Tank, Vertical Slice

library(DBI)
library(RPostgres)
library(dplyr)
library(tidyr)
library(ggplot2)
library(readr)
library(here)
con <- dbConnect(RPostgres::Postgres(), dbname = "nhl", host = "localhost")

games <- read_csv(here("data", "games_raw.csv"))

dbExecute(con, "TRUNCATE games")
## [1] 0
dbWriteTable(con, "games", games, append = TRUE, row.names = FALSE)

dbGetQuery(con, "SELECT season, COUNT(*) AS n FROM games GROUP BY season ORDER BY season")
##     season    n
## 1 20212022 1312
## 2 20222023 1312
## 3 20232024 1312
team_games <- dbGetQuery(con, "
  SELECT
    tg.team,
    tg.season,
    s.season_label,
    tg.game_date,
    tg.points,
    CASE
      WHEN tg.game_date <= s.trade_deadline THEN 'pre'
      ELSE 'post'
    END AS window
  FROM team_games tg
  JOIN season_structure s ON s.season = tg.season
")

nrow(team_games)    # 7872
## [1] 7872
head(team_games)
##   team   season season_label  game_date points window
## 1  ANA 20212022      2021-22 2021-10-13      2    pre
## 2  ANA 20212022      2021-22 2021-10-15      0    pre
## 3  CGY 20212022      2021-22 2021-10-18      1    pre

## 5  WPG 20212022      2021-22 2021-10-21      2    pre

team_summary <- team_games |>
  group_by(team, season, season_label, window) |>
  summarise(
    points_pct = sum(points) / (2 * n()),
    games      = n(),
    .groups    = "drop"
  )

head(team_summary)

## 5 ANA   20232024 2023-24      post        0.263    19
## 6 ANA   20232024 2023-24      pre         0.389    63
wide <- team_summary |>
  select(team, season, season_label, window, points_pct) |>
  pivot_wider(names_from = window, values_from = points_pct) |>
  mutate(drop = post - pre)

ggplot(wide, aes(x = pre, y = drop)) +
  geom_hline(yintercept = 0, linetype = "dashed", color = "gray60") +
  geom_point(alpha = 0.7) +
  geom_smooth(method = "lm", se = TRUE) +
  facet_wrap(~ season_label) +
  labs(
    x = "Pre-deadline points %",
    y = "Drop = post − pre points %",
    title    = "Does late-season performance drop more for bad teams?",
    subtitle = "One dot per team-season. Positive slope = bad teams drop more (tank signal)."
  ) +
  theme_minimal()

fit <- lm(drop ~ pre, data = wide)
summary(fit)
